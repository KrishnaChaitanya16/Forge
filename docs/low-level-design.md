## 2. Environment CRD — Full Schema

### Go types (`pkg/api/v1alpha1/types.go`)

```go
package v1alpha1

import metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"

// +kubebuilder:object:root=true
// +kubebuilder:resource:scope=Cluster,shortName=env,categories=forge
// +kubebuilder:subresource:status
// +kubebuilder:printcolumn:name="State",type=string,JSONPath=`.status.lifecycleState`
// +kubebuilder:printcolumn:name="PR",type=integer,JSONPath=`.spec.pullRequest`
// +kubebuilder:printcolumn:name="URL",type=string,JSONPath=`.status.previewURL`
// +kubebuilder:printcolumn:name="Age",type=date,JSONPath=`.metadata.creationTimestamp`
type Environment struct {
    metav1.TypeMeta   `json:",inline"`
    metav1.ObjectMeta `json:"metadata,omitempty"`
    Spec              EnvironmentSpec   `json:"spec"`
    Status            EnvironmentStatus `json:"status,omitempty"`
}

type EnvironmentSpec struct {
    // +kubebuilder:validation:MaxLength=63
    // +kubebuilder:validation:Pattern=`^[a-z0-9][a-z0-9-]*[a-z0-9]$`
    Repository string `json:"repository"`

    // +kubebuilder:validation:Minimum=1
    PullRequest int `json:"pullRequest"`

    Image     ImageSpec       `json:"image"`
    Resources ResourceSpec    `json:"resources,omitempty"`
    Lifecycle LifecyclePolicy `json:"lifecycle,omitempty"`
    Storage   StorageSpec     `json:"storage,omitempty"`
}

type ImageSpec struct {
    // e.g. "payments-service"
    Repository string `json:"repository"`
    // e.g. "pr-101"
    Tag string `json:"tag"`
    // Populated by operator after first successful pull. sha256:...
    // +kubebuilder:validation:Pattern=`^sha256:[a-f0-9]{64}$`
    Digest string `json:"digest,omitempty"`
}

type LifecyclePolicy struct {
    // Duration string: "15m", "1h", etc.
    // +kubebuilder:validation:Pattern=`^[0-9]+(m|h)$`
    // +kubebuilder:default="15m"
    InactivityTimeout string `json:"inactivityTimeout,omitempty"`

    // +kubebuilder:validation:Pattern=`^[0-9]+(m|h)$`
    // +kubebuilder:default="15m"
    WarmHibernateAfter string `json:"warmHibernateAfter,omitempty"`

    // +kubebuilder:validation:Pattern=`^[0-9]+(m|h)$`
    // +kubebuilder:default="2h"
    DeepHibernateAfter string `json:"deepHibernateAfter,omitempty"`

    // +kubebuilder:validation:Pattern=`^[0-9]+(h|d)$`
    // +kubebuilder:default="72h"
    TTL string `json:"ttl,omitempty"`

    // When true, inactivity-based hibernation is skipped. TTL still applies.
    Pinned bool `json:"pinned,omitempty"`
}

type ResourceSpec struct {
    // +kubebuilder:default="100m"
    CPURequest string `json:"cpuRequest,omitempty"`
    // +kubebuilder:default="500m"
    CPULimit string `json:"cpuLimit,omitempty"`
    // +kubebuilder:default="128Mi"
    MemoryRequest string `json:"memoryRequest,omitempty"`
    // +kubebuilder:default="512Mi"
    MemoryLimit string `json:"memoryLimit,omitempty"`
}

type StorageSpec struct {
    // If false (default), no PVC is created. App gets ephemeral storage only.
    Persistent bool `json:"persistent,omitempty"`
    // Only used when Persistent = true
    // +kubebuilder:default="1Gi"
    SizeLimit string `json:"sizeLimit,omitempty"`
}

// EnvironmentStatus is set exclusively by the operator. Never set by external callers.
type EnvironmentStatus struct {
    // +kubebuilder:validation:Enum=PROVISIONING;ACTIVE;WARM_HIBERNATED;DEEP_HIBERNATED;RESTORING;TERMINATING;TERMINATED;PROVISIONING_FAILED;RESTORATION_FAILED
    LifecycleState LifecycleState `json:"lifecycleState,omitempty"`

    // e.g. "pr-101.dev.local"
    PreviewURL string `json:"previewURL,omitempty"`

    // Name of the managed namespace, e.g. "forge-pr-101"
    NamespaceName string `json:"namespaceName,omitempty"`

    // Last HTTP request timestamp recorded by the gateway
    LastActivityAt *metav1.Time `json:"lastActivityAt,omitempty"`

    // When the environment last finished restoring. Used for grace period.
    RestoredAt *metav1.Time `json:"restoredAt,omitempty"`

    // When the environment first became ACTIVE
    ActiveSince *metav1.Time `json:"activeSince,omitempty"`

    // Absolute expiry time, computed from spec.lifecycle.ttl at creation
    ExpiresAt *metav1.Time `json:"expiresAt,omitempty"`

    // Image digest captured after first successful deployment
    ImageDigest string `json:"imageDigest,omitempty"`

    // Human-readable status message
    Message string `json:"message,omitempty"`

    // Last generation the operator successfully reconciled
    ObservedGeneration int64 `json:"observedGeneration,omitempty"`

    // Standard Kubernetes condition list
    Conditions []metav1.Condition `json:"conditions,omitempty"`
}

type LifecycleState string

const (
    StateProvisioning       LifecycleState = "PROVISIONING"
    StateActive             LifecycleState = "ACTIVE"
    StateWarmHibernated     LifecycleState = "WARM_HIBERNATED"
    StateDeepHibernated     LifecycleState = "DEEP_HIBERNATED"
    StateRestoring          LifecycleState = "RESTORING"
    StateTerminating        LifecycleState = "TERMINATING"
    StateTerminated         LifecycleState = "TERMINATED"
    StateProvisioningFailed LifecycleState = "PROVISIONING_FAILED"
    StateRestorationFailed  LifecycleState = "RESTORATION_FAILED"
)

const (
    ForgeFinalizer = "forge.dev/cleanup"
    ForgeLabel     = "forge.dev/environment"
    GracePeriod    = 5 * time.Minute
)

// +kubebuilder:object:root=true
type EnvironmentList struct {
    metav1.TypeMeta `json:",inline"`
    metav1.ListMeta `json:"metadata,omitempty"`
    Items           []Environment `json:"items"`
}
```

### Reconciler dispatch (`pkg/controller/environment_controller.go`)

```go
const forgeFinalizer = "forge.dev/cleanup"

func (r *EnvironmentReconciler) Reconcile(ctx context.Context, req ctrl.Request) (ctrl.Result, error) {
    log := log.FromContext(ctx)

    env := &v1alpha1.Environment{}
    if err := r.Get(ctx, req.NamespacedName, env); err != nil {
        return ctrl.Result{}, client.IgnoreNotFound(err)
    }

    // Being deleted — run cleanup
    if !env.DeletionTimestamp.IsZero() {
        return r.reconcileTerminating(ctx, env)
    }

    // Ensure finalizer is present
    if !controllerutil.ContainsFinalizer(env, forgeFinalizer) {
        controllerutil.AddFinalizer(env, forgeFinalizer)
        return ctrl.Result{}, r.Update(ctx, env)
    }

    // Dispatch by state
    switch env.Status.LifecycleState {
    case "", v1alpha1.StateProvisioning:
        return r.reconcileProvisioning(ctx, env)
    case v1alpha1.StateActive:
        return r.reconcileActive(ctx, env)
    case v1alpha1.StateWarmHibernated:
        return r.reconcileWarmHibernated(ctx, env)
    case v1alpha1.StateDeepHibernated:
        return r.reconcileDeepHibernated(ctx, env)
    case v1alpha1.StateRestoring:
        return r.reconcileRestoring(ctx, env)
    case v1alpha1.StateTerminating:
        return r.reconcileTerminating(ctx, env)
    default:
        log.Info("unhandled state", "state", env.Status.LifecycleState)
        return ctrl.Result{}, nil
    }
}

// SetupWithManager registers watches
func (r *EnvironmentReconciler) SetupWithManager(mgr ctrl.Manager) error {
    return ctrl.NewControllerManagedBy(mgr).
        For(&v1alpha1.Environment{}).
        // Also reconcile when owned Namespaces change (e.g. deleted externally)
        Owns(&corev1.Namespace{}).
        Complete(r)
}
```

### Provisioning handler (`pkg/controller/reconcile_provisioning.go`)

```go
func (r *EnvironmentReconciler) reconcileProvisioning(ctx context.Context, env *v1alpha1.Environment) (ctrl.Result, error) {
    // 1. Check capacity (delegated to capacity worker via channel)
    if err := r.CapacityWorker.Admit(ctx, env.Name); err != nil {
        // Queued — requeue when a slot opens
        return ctrl.Result{RequeueAfter: 30 * time.Second}, r.setStatus(ctx, env, v1alpha1.StateProvisioning, "waiting for capacity slot")
    }

    // 2. Create namespace (idempotent)
    ns, err := r.Provisioner.EnsureNamespace(ctx, env)
    if err != nil {
        return ctrl.Result{}, r.setFailure(ctx, env, v1alpha1.StateProvisioningFailed, err)
    }

    // 3. Apply ResourceQuota + LimitRange + NetworkPolicy (idempotent, parallel)
    g, gctx := errgroup.WithContext(ctx)
    g.Go(func() error { return r.Provisioner.EnsureResourceQuota(gctx, env, ns) })
    g.Go(func() error { return r.Provisioner.EnsureNetworkPolicies(gctx, env, ns) })
    if err := g.Wait(); err != nil {
        return ctrl.Result{}, r.setFailure(ctx, env, v1alpha1.StateProvisioningFailed, err)
    }

    // 4. Create Deployment
    if err := r.Provisioner.EnsureDeployment(ctx, env, ns, 1); err != nil {
        return ctrl.Result{}, r.setFailure(ctx, env, v1alpha1.StateProvisioningFailed, err)
    }

    // 5. Create Service
    if err := r.Provisioner.EnsureService(ctx, env, ns); err != nil {
        return ctrl.Result{}, r.setFailure(ctx, env, v1alpha1.StateProvisioningFailed, err)
    }

    // 6. Check readiness — requeue if not yet ready
    ready, err := r.Provisioner.IsDeploymentReady(ctx, env, ns)
    if err != nil {
        return ctrl.Result{}, err
    }
    if !ready {
        return ctrl.Result{RequeueAfter: 5 * time.Second}, nil
    }

    // 7. Compute expiry, set ACTIVE
    expiresAt := metav1.NewTime(time.Now().Add(parseDuration(env.Spec.Lifecycle.TTL)))
    env.Status.ExpiresAt     = &expiresAt
    env.Status.PreviewURL    = fmt.Sprintf("%s.dev.local", env.Name)
    env.Status.NamespaceName = ns.Name
    now := metav1.Now()
    env.Status.ActiveSince   = &now
    env.Status.LastActivityAt = &now

    return ctrl.Result{}, r.setStatus(ctx, env, v1alpha1.StateActive, "environment ready")
}
```

### Warm hibernation handler (`pkg/controller/reconcile_hibernation.go`)

```go
func (r *EnvironmentReconciler) reconcileWarmHibernated(ctx context.Context, env *v1alpha1.Environment) (ctrl.Result, error) {
    // Scale Deployment to 0
    if err := r.Provisioner.EnsureDeployment(ctx, env, nil, 0); err != nil {
        return ctrl.Result{}, err
    }
    metrics.RecordHibernation("warm")
    return ctrl.Result{}, r.setStatus(ctx, env, v1alpha1.StateWarmHibernated, "scaled to zero")
}

func (r *EnvironmentReconciler) reconcileDeepHibernated(ctx context.Context, env *v1alpha1.Environment) (ctrl.Result, error) {
    ns := env.Status.NamespaceName

    // Delete Deployment and Service in parallel (namespace + quota retained)
    g, gctx := errgroup.WithContext(ctx)
    g.Go(func() error { return r.Provisioner.DeleteDeployment(gctx, env, ns) })
    g.Go(func() error { return r.Provisioner.DeleteService(gctx, env, ns) })
    if err := g.Wait(); err != nil {
        return ctrl.Result{}, err
    }
    metrics.RecordHibernation("deep")
    return ctrl.Result{}, r.setStatus(ctx, env, v1alpha1.StateDeepHibernated, "deployment and service removed; metadata retained")
}
```

### Restoration handler (`pkg/controller/reconcile_restoring.go`)

```go
func (r *EnvironmentReconciler) reconcileRestoring(ctx context.Context, env *v1alpha1.Environment) (ctrl.Result, error) {
    ns := env.Status.NamespaceName

    // Recreate resources in parallel where possible
    g, gctx := errgroup.WithContext(ctx)
    g.Go(func() error { return r.Provisioner.EnsureDeployment(gctx, env, nil, 1) })
    g.Go(func() error { return r.Provisioner.EnsureService(gctx, env, nil) })
    if err := g.Wait(); err != nil {
        return ctrl.Result{}, err
    }

    // Check readiness — requeue if not yet ready
    ready, err := r.Provisioner.IsDeploymentReady(ctx, env, nil)
    if err != nil {
        return ctrl.Result{}, err
    }
    if !ready {
        // Check timeout
        if time.Since(env.Status.RestoredAt.Time) > 60*time.Second {
            return ctrl.Result{}, r.setStatus(ctx, env, v1alpha1.StateRestorationFailed, "timed out waiting for readiness")
        }
        return ctrl.Result{RequeueAfter: 5 * time.Second}, nil
    }

    now := metav1.Now()
    env.Status.RestoredAt    = &now
    env.Status.LastActivityAt = &now  // reset — prevents immediate re-hibernation

    metrics.RecordRestoration(env.Name, time.Since(env.Status.RestoredAt.Time))
    return ctrl.Result{}, r.setStatus(ctx, env, v1alpha1.StateActive, "restored")
}
```

### Terminating handler (`pkg/controller/reconcile_terminating.go`)

```go
func (r *EnvironmentReconciler) reconcileTerminating(ctx context.Context, env *v1alpha1.Environment) (ctrl.Result, error) {
    ns := env.Status.NamespaceName

    if ns != "" {
        // Deleting the namespace cascades to all resources inside it
        if err := r.Provisioner.DeleteNamespace(ctx, ns); client.IgnoreNotFound(err) != nil {
            return ctrl.Result{}, err
        }
    }

    // Remove finalizer so Kubernetes deletes the CRD object
    controllerutil.RemoveFinalizer(env, forgeFinalizer)
    return ctrl.Result{}, r.Update(ctx, env)
}
```

## 4. Lifecycle Manager — Goroutine Design

### Overview

The lifecycle manager runs a single goroutine that ticks every 60 seconds. It lists all Environment CRDs, evaluates inactivity and TTL policies, and patches the `lifecycleState` field to trigger the next state transition.

```go
// pkg/lifecycle/manager.go

type Manager struct {
    client         client.Client
    capacityWorker *capacity.Worker
    interval       time.Duration // default 60s
}

func (m *Manager) Run(ctx context.Context) {
    ticker := time.NewTicker(m.interval)
    defer ticker.Stop()
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            m.evaluate(ctx)
        }
    }
}

func (m *Manager) evaluate(ctx context.Context) {
    list := &v1alpha1.EnvironmentList{}
    if err := m.client.List(ctx, list); err != nil {
        return
    }
    for i := range list.Items {
        env := &list.Items[i]
        m.evaluateOne(ctx, env)
    }
}

func (m *Manager) evaluateOne(ctx context.Context, env *v1alpha1.Environment) {
    now := time.Now()

    // Hard TTL — overrides everything including pin
    if env.Status.ExpiresAt != nil && now.After(env.Status.ExpiresAt.Time) {
        m.patchState(ctx, env, v1alpha1.StateTerminating, "TTL expired")
        return
    }

    // Only evaluate inactivity for ACTIVE environments
    if env.Status.LifecycleState != v1alpha1.StateActive {
        return
    }

    // Skip if pinned
    if env.Spec.Lifecycle.Pinned {
        return
    }

    // Skip if within grace period after last restore
    if env.Status.RestoredAt != nil && now.Sub(env.Status.RestoredAt.Time) < v1alpha1.GracePeriod {
        return
    }

    if env.Status.LastActivityAt == nil {
        return
    }

    idle := now.Sub(env.Status.LastActivityAt.Time)
    deepAfter := parseDuration(env.Spec.Lifecycle.DeepHibernateAfter)
    warmAfter  := parseDuration(env.Spec.Lifecycle.WarmHibernateAfter)

    switch {
    case idle > deepAfter:
        m.patchState(ctx, env, v1alpha1.StateDeepHibernated, "idle beyond deep-hibernate threshold")
    case idle > warmAfter:
        m.patchState(ctx, env, v1alpha1.StateWarmHibernated, "idle beyond warm-hibernate threshold")
    }
}

func (m *Manager) patchState(ctx context.Context, env *v1alpha1.Environment, state v1alpha1.LifecycleState, msg string) {
    patch := client.MergeFrom(env.DeepCopy())
    env.Status.LifecycleState = state
    env.Status.Message        = msg
    m.client.Status().Patch(ctx, env, patch)
}
```

### Why patch, not update

`Patch` (MergeFrom) sends only the diff, not the whole object. This avoids race conditions where the operator has also updated the status between the List and the Patch — the patch only changes the two fields it intends to, rather than overwriting the whole status with a potentially stale copy.

## 5. Provisioner — Resource Creation Order

### Namespace provisioner (`pkg/provisioner/namespace.go`)

```go
func (p *Provisioner) EnsureNamespace(ctx context.Context, env *v1alpha1.Environment) (*corev1.Namespace, error) {
    name := "forge-" + env.Name
    ns := &corev1.Namespace{
        ObjectMeta: metav1.ObjectMeta{
            Name: name,
            Labels: map[string]string{
                "forge.dev/environment": env.Name,
                "forge.dev/managed":    "true",
            },
            // Cluster-scoped Environment CRD owns the Namespace
            OwnerReferences: []metav1.OwnerReference{
                *metav1.NewControllerRef(env, v1alpha1.GroupVersionKind),
            },
        },
    }
    // CreateOrUpdate is idempotent
    _, err := ctrl.CreateOrUpdate(ctx, p.client, ns, func() error { return nil })
    return ns, err
}
```

### NetworkPolicy provisioner (`pkg/provisioner/namespace.go`)

```go
func (p *Provisioner) EnsureNetworkPolicies(ctx context.Context, env *v1alpha1.Environment, ns *corev1.Namespace) error {
    policies := []*networkingv1.NetworkPolicy{
        defaultDenyPolicy(ns.Name),
        allowGatewayIngressPolicy(ns.Name, p.gatewayNamespace),
        allowDNSEgressPolicy(ns.Name),
    }
    for _, pol := range policies {
        if _, err := ctrl.CreateOrUpdate(ctx, p.client, pol, func() error { return nil }); err != nil {
            return err
        }
    }
    return nil
}

func allowGatewayIngressPolicy(nsName, gatewayNS string) *networkingv1.NetworkPolicy {
    return &networkingv1.NetworkPolicy{
        ObjectMeta: metav1.ObjectMeta{Name: "allow-gateway-ingress", Namespace: nsName},
        Spec: networkingv1.NetworkPolicySpec{
            PodSelector: metav1.LabelSelector{},
            Ingress: []networkingv1.NetworkPolicyIngressRule{{
                From: []networkingv1.NetworkPolicyPeer{{
                    NamespaceSelector: &metav1.LabelSelector{
                        MatchLabels: map[string]string{"kubernetes.io/metadata.name": gatewayNS},
                    },
                }},
            }},
            PolicyTypes: []networkingv1.PolicyType{networkingv1.PolicyTypeIngress},
        },
    }
}

func allowDNSEgressPolicy(nsName string) *networkingv1.NetworkPolicy {
    port53UDP := intstr.FromInt(53)
    port53TCP := intstr.FromInt(53)
    protoUDP  := corev1.ProtocolUDP
    protoTCP  := corev1.ProtocolTCP
    return &networkingv1.NetworkPolicy{
        ObjectMeta: metav1.ObjectMeta{Name: "allow-dns-egress", Namespace: nsName},
        Spec: networkingv1.NetworkPolicySpec{
            PodSelector: metav1.LabelSelector{},
            Egress: []networkingv1.NetworkPolicyEgressRule{{
```

### Wake lock (`pkg/gateway/wakelock.go`)

```go
// WakeLock ensures exactly one restoration per environment regardless of concurrent requests.
type WakeLock struct {
    mu      sync.Mutex
    entries map[string]*wakeEntry
}

type wakeEntry struct {
    done chan struct{} // closed when restore completes (success or failure)
    err  error
}

func NewWakeLock() *WakeLock {
    return &WakeLock{entries: make(map[string]*wakeEntry)}
}

// Wake calls triggerFn exactly once for envName.
// Concurrent callers wait until the first triggerFn completes.
func (w *WakeLock) Wake(ctx context.Context, envName string, triggerFn func(context.Context) error) error {
    w.mu.Lock()
    if entry, exists := w.entries[envName]; exists {
        // Another goroutine is already restoring — wait for it
        w.mu.Unlock()
        select {
        case <-entry.done:
            return entry.err
        case <-ctx.Done():
            return ctx.Err()
        }
    }

    // First goroutine — own the restore
    entry := &wakeEntry{done: make(chan struct{})}
    w.entries[envName] = entry
    w.mu.Unlock()

    entry.err = triggerFn(ctx)

    // Notify all waiters
    w.mu.Lock()
    delete(w.entries, envName)
    w.mu.Unlock()
    close(entry.done)

    return entry.err
}
```

### HTTP handler chain (`pkg/gateway/handler.go`)

```go
type Handler struct {
    envCache   cache.Cache
    forgeAPI   *http.Client   // client to POST /api/v1/environments/:name/restore
    forgeAPIURL string
    wakeLock   *WakeLock
    timeout    time.Duration  // default 60s
}

func (h *Handler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    envName := hostnameToEnvName(r.Host) // "pr-101.dev.local" → "pr-101"
    if envName == "" {
        http.Error(w, "unknown host", http.StatusNotFound)
        return
    }

    state, err := GetEnvironmentState(h.envCache, envName)
    if errors.IsNotFound(err) {
        http.Error(w, "environment not found", http.StatusNotFound)
        return
    }

    switch state {
    case v1alpha1.StateActive:
        h.forwardToApp(w, r, envName)

    case v1alpha1.StateWarmHibernated, v1alpha1.StateDeepHibernated:
        h.wakeAndForward(w, r, envName)

    case v1alpha1.StateProvisioning, v1alpha1.StateRestoring:
        h.serveWakingPage(w, r, "Environment is starting up…")

    case v1alpha1.StateTerminating, v1alpha1.StateTerminated:
        http.Error(w, "environment has been terminated", http.StatusGone)

    default:
        http.Error(w, "environment unavailable", http.StatusServiceUnavailable)
    }
}

func (h *Handler) wakeAndForward(w http.ResponseWriter, r *http.Request, envName string) {
    ctx, cancel := context.WithTimeout(r.Context(), h.timeout)
    defer cancel()

    err := h.wakeLock.Wake(ctx, envName, func(ctx context.Context) error {
        return h.triggerRestore(ctx, envName)
    })

    if err != nil {
        if isAcceptHTML(r) {
            h.serveErrorPage(w, "Wake-up timed out. Try refreshing.")
        } else {
            http.Error(w, "environment restoration timed out", http.StatusServiceUnavailable)
        }
        return
    }

    h.forwardToApp(w, r, envName)
}

func (h *Handler) triggerRestore(ctx context.Context, envName string) error {
    // POST /api/v1/environments/:name/restore
    req, _ := http.NewRequestWithContext(ctx, http.MethodPost,
        h.forgeAPIURL+"/api/v1/environments/"+envName+"/restore", nil)
    resp, err := h.forgeAPI.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()

    // Poll until ACTIVE or timeout (ctx handles the timeout)
    for {
        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-time.After(2 * time.Second):
        }
        state, err := GetEnvironmentState(h.envCache, envName)
        if err != nil {
            continue
        }
        if state == v1alpha1.StateActive {
            return nil
        }
        if state == v1alpha1.StateRestorationFailed {
            return fmt.Errorf("restoration failed")
        }
    }
}

// serveWakingPage returns an HTML page with a meta-refresh for browser clients
func (h *Handler) serveWakingPage(w http.ResponseWriter, r *http.Request, msg string) {
    if !isAcceptHTML(r) {
        http.Error(w, msg, http.StatusAccepted)
        return
    }
    w.Header().Set("Content-Type", "text/html")
    w.WriteHeader(http.StatusAccepted)
    fmt.Fprintf(w, `<!DOCTYPE html><html><head>
<meta http-equiv="refresh" content="3">
<title>Forge — Waking up</title></head><body>
<h2>⏳ %s</h2><p>This page will refresh automatically.</p>
</body></html>`, msg)
}

func isAcceptHTML(r *http.Request) bool {
    return strings.Contains(r.Header.Get("Accept"), "text/html")
}
```

```go
func bearerTokenMiddleware(next http.Handler, token string) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        auth := r.Header.Get("Authorization")
        if !strings.HasPrefix(auth, "Bearer ") || strings.TrimPrefix(auth, "Bearer ") != token {
            http.Error(w, "unauthorized", http.StatusUnauthorized)
            return
        }
        next.ServeHTTP(w, r)
    })
}
```

## 8. Webhook Receiver — Event Processing

### Signature validation + dispatch

```go
func (h *WebhookHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    body, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "failed to read body", http.StatusBadRequest)
        return
    }

    // HMAC-SHA256 validation
    sig := r.Header.Get("X-Hub-Signature-256")
    if !validateSignature(body, sig, h.secret) {
        http.Error(w, "invalid signature", http.StatusUnauthorized)
        return
    }

    event := r.Header.Get("X-GitHub-Event")
    switch event {
    case "pull_request":
        h.handlePullRequest(w, r, body)
    default:
        w.WriteHeader(http.StatusNoContent) // ignore other events
    }
}

func (h *WebhookHandler) handlePullRequest(w http.ResponseWriter, r *http.Request, body []byte) {
    var payload GitHubPRPayload
    if err := json.Unmarshal(body, &payload); err != nil {
        http.Error(w, "invalid payload", http.StatusBadRequest)
        return
    }

    envName := fmt.Sprintf("pr-%d", payload.PullRequest.Number)

    switch payload.Action {
    case "opened", "reopened":
        h.createOrUpdateEnvironment(r.Context(), envName, payload)
    case "synchronize":
        h.updateImageTag(r.Context(), envName, payload)
    case "closed":
        h.terminateEnvironment(r.Context(), envName)
    }

    w.WriteHeader(http.StatusNoContent)
}

func validateSignature(body []byte, sig, secret string) bool {
    mac := hmac.New(sha256.New, []byte(secret))
    mac.Write(body)
    expected := "sha256=" + hex.EncodeToString(mac.Sum(nil))
    return hmac.Equal([]byte(expected), []byte(sig))
}
```

## 9. Capacity Worker — Serialized Admission

### Design

All admission decisions (provision, wake at capacity) go through a single goroutine. This prevents two simultaneous requests from both seeing "9 active environments" and both being admitted, resulting in 11.

```go
type admitRequest struct {
    envName string
    reply   chan error
}

type Worker struct {
    client       client.Client
    maxActive    int
    admitCh      chan admitRequest
    releaseCh    chan string  // env name released (hibernated or terminated)
}

func (w *Worker) Run(ctx context.Context) {
    for {
        select {
        case <-ctx.Done():
            return
        case req := <-w.admitCh:
            req.reply <- w.admitOne(ctx, req.envName)
        case name := <-w.releaseCh:
            _ = name // capacity freed — drain the pending queue
            w.drainQueue(ctx)
        }
    }
}

func (w *Worker) admitOne(ctx context.Context, envName string) error {
    active, err := w.countActive(ctx)
    if err != nil {
        return err
    }
    if active < w.maxActive {
        return nil // slot available
    }
    // Try to evict LRU active environment
    if err := w.evictLRU(ctx); err != nil {
        return fmt.Errorf("at capacity (%d active); no evictable environment: %w", w.maxActive, err)
    }
    return nil
}

// Admit is called by the reconciler (blocks until a slot is available or ctx expires)
func (w *Worker) Admit(ctx context.Context, envName string) error {
    reply := make(chan error, 1)
    select {
    case w.admitCh <- admitRequest{envName: envName, reply: reply}:
    case <-ctx.Done():
        return ctx.Err()
    }
    select {
    case err := <-reply:
        return err
    case <-ctx.Done():
        return ctx.Err()
    }
}
```
